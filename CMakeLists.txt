cmake_minimum_required(VERSION 3.20)
project(lab04_sync LANGUAGES CXX)
option(LAB_PORTABLE "Build algorithm validation backend instead of Windows API" OFF)
if(NOT WIN32 AND NOT LAB_PORTABLE)
  message(FATAL_ERROR "Windows API required. For validation only use -DLAB_PORTABLE=ON")
endif()
add_executable(philosophers src/main.cpp)
target_compile_features(philosophers PRIVATE cxx_std_17)
if(LAB_PORTABLE)
  target_compile_definitions(philosophers PRIVATE LAB_PORTABLE=1)
  find_package(Threads REQUIRED)
  target_link_libraries(philosophers PRIVATE Threads::Threads)
endif()
if(MSVC)
  target_compile_options(philosophers PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(philosophers PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  enable_testing()
  add_test(NAME correctness COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_app.py $<TARGET_FILE:philosophers>)
endif()
